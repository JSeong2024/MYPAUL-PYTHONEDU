cmake_minimum_required(VERSION 3.20)
project(ch18_advanced_telemetry LANGUAGES CXX)
find_package(Threads REQUIRED)
add_library(telemetry STATIC src/telemetry.cpp)
target_include_directories(telemetry PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(telemetry PUBLIC cxx_std_20)
set_target_properties(telemetry PROPERTIES CXX_EXTENSIONS OFF)
target_link_libraries(telemetry PUBLIC Threads::Threads)
if(WIN32)
  target_link_libraries(telemetry PUBLIC ws2_32)
endif()
add_executable(telemetry_demo src/main.cpp)
target_link_libraries(telemetry_demo PRIVATE telemetry)
include(CTest)
if(BUILD_TESTING)
  add_executable(telemetry_tests tests/test_telemetry.cpp)
  target_link_libraries(telemetry_tests PRIVATE telemetry)
  add_test(NAME telemetry_checks COMMAND telemetry_tests)
  add_test(NAME telemetry_output COMMAND telemetry_demo)
  set_tests_properties(telemetry_output PROPERTIES PASS_REGULAR_EXPRESSION "Accepted: 3, Rejected: 1")
endif()
foreach(t telemetry telemetry_demo)
  if(MSVC)
    target_compile_options(${t} PRIVATE /W4 /EHsc /permissive- /utf-8)
  else()
    target_compile_options(${t} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
