cmake_minimum_required(VERSION 3.20)
project(ch18_loopback_echo LANGUAGES CXX)
find_package(Threads REQUIRED)
add_library(echo_lib STATIC src/echo.cpp)
target_include_directories(echo_lib PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(echo_lib PUBLIC cxx_std_20)
set_target_properties(echo_lib PROPERTIES CXX_EXTENSIONS OFF)
target_link_libraries(echo_lib PUBLIC Threads::Threads)
if(WIN32)
  target_link_libraries(echo_lib PUBLIC ws2_32)
endif()
add_executable(echo_demo src/main.cpp)
target_link_libraries(echo_demo PRIVATE echo_lib)
include(CTest)
if(BUILD_TESTING)
  add_executable(echo_tests tests/test_echo.cpp)
  target_link_libraries(echo_tests PRIVATE echo_lib)
  add_test(NAME echo_behavior COMMAND echo_tests)
  add_test(NAME echo_output COMMAND echo_demo)
  set_tests_properties(echo_output PROPERTIES PASS_REGULAR_EXPRESSION "Echo: sensor-ready")
endif()
foreach(t echo_lib echo_demo)
  if(MSVC)
    target_compile_options(${t} PRIVATE /W4 /EHsc /permissive- /utf-8)
  else()
    target_compile_options(${t} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
if(TARGET echo_tests)
  if(MSVC)
    target_compile_options(echo_tests PRIVATE /W4 /EHsc /permissive- /utf-8)
  else()
    target_compile_options(echo_tests PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endif()
