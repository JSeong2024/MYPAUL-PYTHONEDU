cmake_minimum_required(VERSION 3.21)
project(QualityGate VERSION 1.0.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
option(CH17_ENABLE_SANITIZERS "Enable ASan and UBSan for local debugging" OFF)
add_library(quality_gate STATIC src/batch.cpp)
add_library(QualityGate::quality_gate ALIAS quality_gate)
target_compile_features(quality_gate PUBLIC cxx_std_20)
target_include_directories(quality_gate PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
set_target_properties(quality_gate PROPERTIES CXX_EXTENSIONS OFF)
add_executable(quality_gate_cli app/main.cpp)
target_link_libraries(quality_gate_cli PRIVATE quality_gate)
add_executable(quality_gate_bench tools/benchmark.cpp)
target_link_libraries(quality_gate_bench PRIVATE quality_gate)
include(CTest)
if(BUILD_TESTING)
 add_executable(quality_gate_tests tests/test_batch.cpp)
 target_link_libraries(quality_gate_tests PRIVATE quality_gate)
 add_executable(quality_gate_properties tests/test_properties.cpp)
 target_link_libraries(quality_gate_properties PRIVATE quality_gate)
 add_test(NAME quality_unit COMMAND quality_gate_tests)
 add_test(NAME quality_properties COMMAND quality_gate_properties)
 add_test(NAME quality_cli COMMAND quality_gate_cli)
 set_tests_properties(quality_cli PROPERTIES PASS_REGULAR_EXPRESSION "Devices: 2")
endif()
foreach(target quality_gate quality_gate_cli quality_gate_bench)
 if(MSVC)
  target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
 else()
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
 endif()
endforeach()
if(TARGET quality_gate_tests)
 foreach(target quality_gate_tests quality_gate_properties)
  if(MSVC)
   target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
  else()
   target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
 endforeach()
endif()
if(CH17_ENABLE_SANITIZERS)
 if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang" AND NOT WIN32)
  foreach(target quality_gate quality_gate_cli quality_gate_bench)
   target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -g)
   target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endforeach()
  if(TARGET quality_gate_tests)
   foreach(target quality_gate_tests quality_gate_properties)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -g)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
   endforeach()
  endif()
 else()
  message(FATAL_ERROR "CH17_ENABLE_SANITIZERS supports GCC/Clang on POSIX in this exercise")
 endif()
endif()
install(TARGETS quality_gate EXPORT QualityGateTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
set(CONFIG_DIR "${CMAKE_INSTALL_LIBDIR}/cmake/QualityGate")
configure_package_config_file(cmake/QualityGateConfig.cmake.in
 "${CMAKE_CURRENT_BINARY_DIR}/QualityGateConfig.cmake"
 INSTALL_DESTINATION "${CONFIG_DIR}")
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/QualityGateConfigVersion.cmake"
 VERSION "${PROJECT_VERSION}" COMPATIBILITY SameMajorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/QualityGateConfig.cmake"
              "${CMAKE_CURRENT_BINARY_DIR}/QualityGateConfigVersion.cmake"
 DESTINATION "${CONFIG_DIR}")
install(EXPORT QualityGateTargets NAMESPACE QualityGate::
 DESTINATION "${CONFIG_DIR}")
