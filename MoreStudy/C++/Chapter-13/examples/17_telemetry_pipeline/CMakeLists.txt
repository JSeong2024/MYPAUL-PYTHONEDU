cmake_minimum_required(VERSION 3.21)
project(telemetry_demo LANGUAGES CXX)

add_library(telemetry_report STATIC src/report.cpp)
target_include_directories(telemetry_report PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(telemetry_report PUBLIC cxx_std_20)
set_target_properties(telemetry_report PROPERTIES CXX_EXTENSIONS OFF)

add_executable(telemetry_demo src/main.cpp)
target_link_libraries(telemetry_demo PRIVATE telemetry_report)

include(CTest)
if(BUILD_TESTING)
    add_executable(test_pipeline tests/test_pipeline.cpp)
    target_link_libraries(test_pipeline PRIVATE telemetry_report)
    add_test(NAME telemetry_pipeline_tests COMMAND test_pipeline)
endif()

foreach(target telemetry_report telemetry_demo test_pipeline)
    if(TARGET ${target})
        if(MSVC)
            target_compile_options(${target} PRIVATE /W4 /EHsc /utf-8)
        else()
            target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
        endif()
    endif()
endforeach()
