cmake_minimum_required(VERSION 3.20)
project(event_summary LANGUAGES CXX)
add_executable(event_summary main.cpp event_summary.cpp)
target_compile_features(event_summary PRIVATE cxx_std_23)
if(MSVC)
    target_compile_options(event_summary PRIVATE /W4 /permissive- /utf-8)
else()
    target_compile_options(event_summary PRIVATE -Wall -Wextra -Wpedantic)
endif()
