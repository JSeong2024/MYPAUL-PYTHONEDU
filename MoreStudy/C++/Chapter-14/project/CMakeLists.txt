cmake_minimum_required(VERSION 3.20)
project(sensor_dispatcher LANGUAGES CXX)

add_library(dispatch_lib STATIC src/dispatcher.cpp)
target_include_directories(dispatch_lib PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(dispatch_lib PUBLIC cxx_std_20)
set_target_properties(dispatch_lib PROPERTIES CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
target_link_libraries(dispatch_lib PUBLIC Threads::Threads)

add_executable(sensor_dispatcher src/main.cpp)
target_link_libraries(sensor_dispatcher PRIVATE dispatch_lib)

include(CTest)
if(BUILD_TESTING)
    add_executable(test_dispatcher tests/test_dispatcher.cpp)
    target_link_libraries(test_dispatcher PRIVATE dispatch_lib)
    add_test(NAME dispatcher_behavior COMMAND test_dispatcher)
    set_tests_properties(dispatcher_behavior PROPERTIES TIMEOUT 10)
    add_test(NAME dispatcher_output COMMAND sensor_dispatcher)
    set_tests_properties(dispatcher_output PROPERTIES PASS_REGULAR_EXPRESSION "Processed: 4" TIMEOUT 10)
endif()

foreach(target dispatch_lib sensor_dispatcher test_dispatcher)
    if(TARGET ${target})
        if(MSVC)
            target_compile_options(${target} PRIVATE /W4 /EHsc /permissive- /utf-8)
        else()
            target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
        endif()
    endif()
endforeach()
