cmake_minimum_required(VERSION 3.25)
project(device_session LANGUAGES CXX)
add_executable(device_session main.cpp session.cpp)
target_compile_features(device_session PRIVATE cxx_std_20)
set_target_properties(device_session PROPERTIES CXX_EXTENSIONS OFF CXX_STANDARD_REQUIRED ON)
if (MSVC)
    target_compile_options(device_session PRIVATE /W4 /permissive- /utf-8)
else()
    target_compile_options(device_session PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME device_session_demo COMMAND device_session)
set_tests_properties(device_session_demo PROPERTIES
    PASS_REGULAR_EXPRESSION "Closed sessions: 1")
